# 06. 예제 코드 모음 — 가장 작은 sgf 게임

> 이 문서의 코드는 실제 sgfr 튜토리얼(`Projects/sgfr/Sources/sgfr/Tutorial/`)에서 사용된 패턴을 정리한 것입니다.
> 전부 컴파일되는 완전한 코드가 아니라 **핵심 골격**입니다. 실제 동작 코드는 해당 튜토리얼을 참고하세요.

## 예제 1. 가장 작은 2D 씬 — 사각형 하나

```cpp
// Scene2D를 상속받고 OnRender에서 그리기만 하면 된다.
class MyScene : public Scene2D
{
public:
	void OnEnter() override
	{
		// 카메라: 뷰포트 크기(화면 좌표 = 픽셀)로 직교 투영 설정
		GetCamera2D()->SetOrthographic2D(1280, 720);
	}

	void OnRender() override
	{
		// 즉시 그리기: 중앙(400,300)에 200x150 빨간 사각형
		g_cRenderer2D.DrawRect(vec2(400, 300), vec2(200, 150), color(255, 0, 0));
	}
};
```

**최소 실행** (sgfr의 29번이 실제로 이렇게 합니다):

```cpp
int main()
{
	// 1. 윈도우/디바이스/렌더러/리소스 전부 초기화
	if (!g_cApp.Initialize(L"My Game", 1280, 720))
		return -1;

	// 2. 첫 씬 시작 (ApplicationDidFinishLaunching 안에서 하는 게 정석)
	g_cDirector.RunScene(new MyScene());

	// 3. 메인 루프 (메시지 → 타이머 → 입력 → 갱신 → 렌더)
	g_cApp.Run();
	return 0;
}
```

## 예제 2. 스태틱 선언 — 1회 bake, id로 그리기

```cpp
class MyScene : public Scene2D
{
public:
	void OnEnter() override
	{
		GetCamera2D()->SetOrthographic2D(1280, 720);

		// OnEnter에서 1회 선언 → staticId 반환
		floorId_  = DeclareStaticRect(rect(0, 0, 1280, 100), color(40, 40, 60));
		boxId_    = DeclareStaticRect(rect(100, 200, 80, 80),  color(255, 200, 0));
		circleId_ = DeclareStaticCircle(vec2(600, 400), 50, color(100, 200, 255), 32);
	}

	void OnRender() override
	{
		// 매 프레임: id만 예약 → 렌더러가 캐시된 버텍스로 전송 (재계산 없음)
		RenderStatic(floorId_);
		RenderStatic(boxId_);
		RenderStatic(circleId_);
	}

private:
	_u64 floorId_ = 0;
	_u64 boxId_ = 0;
	_u64 circleId_ = 0;
};
```

## 예제 3. Shape2D — 정적 2D 도형 노드 (1회 bake, 자동 드로우)

```cpp
class MyScene : public Scene2D
{
public:
	void OnEnter() override
	{
		GetCamera2D()->SetOrthographic2D(1280, 720);

		// AddChild 전에 도형 설정 → AddChild 직후 OnEnter가 1회 bake, 이후 매 프레임 자동 드로우
		Shape2D* pPanel = dbg_new Shape2D("패널");
		pPanel->SetRect(rect(0, 0, 1280, 100), color(40, 40, 60));
		AddChild(pPanel, 0);

		Shape2D* pBadge = dbg_new Shape2D("뱃지");
		pBadge->SetCircle(vec2(600, 400), 50, color(100, 200, 255), 32);
		AddChild(pBadge, 1);               // zOrder: 패널 위

		// 테두리 선: 절대좌표 고정 — 가만히 있는 UI에 적합
		Shape2D* pBorder = dbg_new Shape2D("테두리");
		pBorder->SetLine(vec2(10, 10), vec2(200, 10), 3.0f, color::WHITE);
		AddChild(pBorder, 2);
	}
};
```

> Shape2D는 **완전 정적(절대좌표 1회 bake)** — 움직이면 안 되는 UI/테두리/배경에 쓰세요.
> `staticId_`는 OnExit에서 무효화만 됩니다 (staticCache 순수 증가 — 반납 없음).

## 예제 4. 움직이는 도형 — 2D 프리미티브 메시 + Transform

```cpp
class Player : public GameObject
{
public:
	Player()
	{
		// 2D 프리미티브 메시 (디폴트 리소스 — 단위 크기, VertexPTC)
		SetMesh((Mesh*)g_cResourceMgr.Find(
			g_cResourceMgr.GetPrimitiveMesh2DKey(PrimitiveMesh2DType::Rect)));

		// 게임오브젝트가 기본 제공하는 머티리얼 — 채움색은 baseColor 틴트로
		GetMaterial()->SetBaseColor(color(255, 200, 0));
	}

	void OnUpdate(const jc::TimeSpan& _dt) override   // 씬이 트리를 순회하며 호출
	{
		// 움직임 = Transform 이동 (월드 행렬은 자동 계산 — 부모 변환 포함)
		GetTransform()->SetLocalPosition(GetTransform()->GetLocalPosition() + vec3(1, 0, 0));
	}
};

// 씬에서는 AddChild만 하면 끝 — 그리기 코드가 없다.
pPlayer_ = new Player();
AddChild(pPlayer_, 1);
```

> **권장 분기**: 가만히 있는 도형 = Shape2D(예제 3), 움직이는 도형 = SetMesh(이 예제),
> 모양이 매 프레임 변하는 채움(게이지 등) = RenderDynamic. 스태틱 bake는 런타임 재계산이 없어 가장 빠릅니다.

## 예제 5. 게임 오브젝트 트리 + 움직이는 캐릭터

```cpp
class Player : public GameObject
{
public:
	// OnRender에서 즉시 그리기 방식 (예제 4의 SetMesh 자동 드로우와 같은 결과 — 선택)
	// (게임 오브젝트는 트리 노드 — 그리기는 OnRender, 갱신은 씬/자신이 구성)
	void OnRender() override
	{
		// 자신의 Transform 위치로 즉시 그리기 (렌더러 직접 호출)
		const vec3 pos = GetTransform()->GetLocalPosition();
		g_cRenderer2D.DrawRect(vec2(pos.x, pos.y), vec2(64, 64), color(0, 200, 100));
	}
};

class MyScene : public Scene2D
{
public:
	void OnEnter() override
	{
		GetCamera2D()->SetOrthographic2D(1280, 720);

		pPlayer_ = new Player();
		pPlayer_->SetName("Player");
		AddChild(pPlayer_, 1);          // zOrder 1 — 씬 트리에 등록 (OnEnter 1회 호출)
	}

	void OnUpdate(const jc::TimeSpan& _dt) override
	{
		const _f32 dt = static_cast<_f32>(_dt.GetTotalSeconds());

		// 방향키 입력 (폴링) — Transform으로 이동
		const _f32 speed = 200.0f;
		vec3 move;
		if (g_cInput.IsKeyDown(VK_LEFT))  move.x -= speed * dt;
		if (g_cInput.IsKeyDown(VK_RIGHT)) move.x += speed * dt;
		if (g_cInput.IsKeyDown(VK_UP))    move.y += speed * dt;
		if (g_cInput.IsKeyDown(VK_DOWN))  move.y -= speed * dt;

		pPlayer_->GetTransform()->SetLocalPosition(
			pPlayer_->GetTransform()->GetLocalPosition() + move);
	}

private:
	Player* pPlayer_ = nullptr;
};
```

## 예제 6. 3D 큐브 + 카메라 궤도

```cpp
class CubeScene : public Scene3D
{
public:
	void OnEnter() override
	{
		// 카메라: 눈 위치, 목표, 위 방향
		GetCamera3D()->SetLookAt(vec3(3, 2, -3), vec3(0, 0, 0), vec3(0, 1, 0));
		GetCamera3D()->SetPerspectiveDegrees(60.0f, 1280.0f / 720.0f, 0.1f, 100.0f);
	}

	void OnUpdate(const jc::TimeSpan& _dt) override
	{
		// 표준 조작 한 줄: WASD 이동 + 드래그 궤도 + 휠 줌
		GetCamera3D()->DriveDefault3D(g_cInput, _dt);
	}

	void OnRender() override
	{
		// 프리미티브 메시 (디폴트 리소스 — 키로 조회)
		Mesh* pCubeMesh = (Mesh*)g_cResourceMgr.Find(
			g_cResourceMgr.GetPrimitiveMesh3DKey(PrimitiveMesh3DType::Cube));

		// 디폴트 머티리얼 (불투명 + 깊이 읽기/쓰기 + 기본 셰이더)
		Material* pMat = g_cResourceMgr.GetDefaultMaterial3D();

		// 월드 행렬: 스케일 1.5배 + 45도 회전 + 위치 (0, 0.5, 0)
		mat4 world = mat4::CreateScale(1.5f) * mat4::CreateRotationY(jc_math_deg2rad(45.0f))
		           * mat4::CreateTranslation(0, 0.5f, 0);

		DrawMesh(pCubeMesh, pMat, world);

		// 디버그: 바닥 그리드 + 축
		g_cRenderer3D.DrawGrid(10, 1.0f, color(60, 60, 80));
		g_cRenderer3D.DrawAxis(2.0f);
	}
};
```

## 예제 8. 리소스 로드 + 머티리얼 만들기

```cpp
// 텍스처 로드 (경로 중복 로드 방지 — 도서관이 처리)
const _u64 texKey = g_cResourceMgr.LoadTextureFromFile("Assets/player.png");
Texture* pTex = (Texture*)g_cResourceMgr.Find(texKey);

// 2D 기본 머티리얼: 알파 블렌드 + 깊이 꺼짐 + 기본 2D 셰이더 (엔진이 미리 준비)
Material* pMat2D = g_cResourceMgr.GetDefaultMaterial2D();

// 2D 씬에서 이미지로 그리기
g_cRenderer2D.DrawSprite(pTex, vec2(200, 200), vec2(128, 128));   // 즉시 1장
// 또는 스태틱으로: DeclareStaticImage(pTex, rect(100, 100, 128, 128));

// 메시 생성 (사각형 2D 프리미티브 — 디폴트 리소스)
Mesh* pMesh = (Mesh*)g_cResourceMgr.Find(
	g_cResourceMgr.GetPrimitiveMesh2DKey(PrimitiveMesh2DType::Rect));

// 3D처럼 메시+머티리얼로 그리기 (Scene2D::DrawMesh — Renderer3D로 위임)
DrawMesh(pMesh, pMat2D, mat4::CreateTranslation(0, 0, 0));
```

> **머티리얼을 직접 만들려면**: `Material* pMat = new Material();` → 텍스처 슬롯 설정
> (`SetTextureKey(0, texKey)` — `Material.cpp:61`) → `g_cResourceMgr.Add(pMat)`로 소유권을 넘깁니다.
> 상태(블렌드/깊이)는 `g_cDevice.SetBlendMode(bmAlpha)` 같은 디바이스 세터 또는
> PipelineState 객체로 지정합니다.

## 주의 사항 (실전 팁)

1. `OnEnter`에서 리소스/스태틱을 준비하고, `OnExit`에서 정리합니다 (씬 수명주기).
2. 씬에 들어온 뒤에만 `AddChild`를 씁니다 (running 씬 검증).
3. 도형 선택: **가만히 있으면 Shape2D, 움직이면 SetMesh(2D 메시), 매 프레임 변하면 RenderDynamic.**
4. 카메라 조작은 `DriveDefault2D/3D` 한 줄로 시작하고, 나중에 직접 구현합니다.
5. 리소스 키는 꼭 보관하세요. 매번 `LoadTextureFromFile`을 호출해도 중복 로드는 안 되지만 키는 다시 발급됩니다.
